LINE Bot + Google Sheets 智慧客服系統
多工作表分類與雲端網頁管理後台

利用 Google Apps Script 的 HTTP Request 分流特性，在同一組 Web App 網址下實現「Webhook 背景服務（doPost）」與「現代化 SPA 視覺管理後台（doGet）」的雙工運作。支援跨分頁穿透檢索、模糊／精確比對演算法，管理者更可直接在瀏覽器即時維護問答資料庫。

02.

第一步：準備 Google 試算表與欄位規格

在雲端硬碟建立一份試算表（例如：LINE Bot 智慧客服資料庫）。每個業務分頁皆須遵循前三欄標準規範：

欄位 A (關鍵字) 欄位 B (回覆內容) 欄位 C (比對模式)
地址 本店地址：台北市信義區信義路五段7號 (台北101)。 contains
營業時間 營業時間：週一至週五 09:00 - 18:00。 contains
聯絡電話 客服專線：(02) 1234-5678。 exact
🎯 欄位 C 比對模式說明
  • contains (模糊包含)：提問句包含此字詞即觸發（如「請問地址在哪」命中「地址」）。
  • exact (完全相符)：訪客必須百分之百精確輸入相同字詞。
🛡️ 系統保留分頁（黑名單）

若建立名為 設定、日誌、Logs、Config 的分頁，系統搜尋引擎會自動忽略，絕不對訪客暴露內部資料。

03.

第二步：部署進階版 GAS 程式碼 (Code.gs)

  1. 在 Google 試算表上方工具列點選 擴充功能 > Apps Script。
  2. 將下方完整程式碼貼入編輯器中，並於第 11 行填入您的 LINE Channel Access Token，且務必於第 14 行自訂專屬的「管理員密碼 (ADMIN_PASSWORD)」。
  3. 點選右上角藍色按鈕 「部署」 > 「新增部署」。
  4. 左側齒輪選擇 「網頁應用程式 (Web app)」，關鍵權限設定如下：
    • 說明：進階安全防護版 v3.1
    • 執行身分：我 (您的 Google 帳號)
    • 誰可以存取：所有人 (Anyone) (供 LINE 伺服器傳送 Webhook；網頁後台已全面由密碼伺服器端鎖定)
  5. 授權審查：點選「審查權限」 > 選擇帳號 > 點選「進階 (Advanced)」 > 「前往『未命名的專案』(不安全)」 > 「允許」。
  6. 複製產生的 網頁應用程式網址 (Web app URL)（結尾為 /exec）。

// 🛡️ 後台管理員專屬密碼（請務必自行修改，防止未授權人員透過網頁端竄改試算表）

// 排除在客服檢索之外的系統分頁名稱（如留空則所有工作表皆作為分類）
const EXCLUDED_SHEETS = ['設定', '日誌', 'Logs', 'Config'];

// 查無關鍵字時的回覆文字
const DEFAULT_FALLBACK_MESSAGE = '感謝您的提問！目前小幫手找不到相關資訊。\n您可以輸入「地址」、「營業時間」或「退換貨」等關鍵字進行查詢喔！';

// 2. 網頁管理前台入口 (doGet)：提供管理者以瀏覽器維護資料
// =================================================================

function doGet(e) {
  const htmlContent = getAdminHtml();
  return HtmlService.createHtmlOutput(htmlContent)
    .setTitle('LINE Bot 智慧客服 • 雲端資料庫管理後台')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

// =================================================================
// 3. LINE Webhook HTTP POST 入口 (doPost)：處理訪客對話
// =================================================================

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      console.warn('收到無內容的請求 (No postData)');
      return ContentService.createTextOutput(JSON.stringify({ status: 'no_data' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const eventData = JSON.parse(e.postData.contents);
    const events = eventData.events;

    // 容錯：處理 LINE 的驗證 ping 或空陣列事件
    if (!events || events.length === 0) {
      console.log('收到驗證事件 (Verification Ping)');
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'verified' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    for (let i = 0; i < events.length; i++) {
      const currentEvent = events[i];
      if (currentEvent.type === 'message' && currentEvent.message && currentEvent.message.type === 'text') {
        handleTextMessage(currentEvent);
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    console.error('doPost 發生異常: ' + error.toString());
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// =================================================================
// 4. 訊息派送與多工作表檢索演算法
// =================================================================
function handleTextMessage(event) {
  const replyToken = event.replyToken;
  const userMessage = event.message.text ? event.message.text.trim() : '';

  console.log('收到使用者提問: ' + userMessage);

  if (!replyToken || replyToken === '00000000000000000000000000000000' || replyToken === 'ffffffffffffffffffffffffffffffff') {
    console.log('偵測到測試專用 ReplyToken，略過回傳作業');
    return;
  }

  // 跨工作表檢索相應解答
  const matchedAnswer = searchAnswerAcrossAllSheets(userMessage);

  // 透過 LINE Reply API 發送給用戶
  replyMessage(replyToken, matchedAnswer);
}

function searchAnswerAcrossAllSheets(userQuery) {
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  if (!spreadsheet) {
    console.error('無法綁定作用中試算表！');
    return '系統錯誤：無法開啟試算表。';
  }

  const allSheets = spreadsheet.getSheets();
  const query = userQuery.toLowerCase();
  let fallbackCandidate = null;

  for (let s = 0; s < allSheets.length; s++) {
    const sheet = allSheets[s];
    const sheetName = sheet.getName();

    // 略過排除清單中的系統分頁
    if (EXCLUDED_SHEETS.indexOf(sheetName) !== -1) {
      continue;
    }

    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) continue;

    // 讀取 A2:C最底列 (A:關鍵字, B:回覆內容, C:比對模式)
    const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();

    for (let i = 0; i < values.length; i++) {
      const keyword = String(values[i][0] || '').trim().toLowerCase();
      const response = String(values[i][1] || '').trim();
      const matchType = String(values[i][2] || 'contains').trim().toLowerCase();

      if (!keyword || !response) continue;

      if (matchType === 'exact') {
        if (query === keyword) {
          console.log(`[完全相符] 命中分頁 [${sheetName}], 關鍵字: ${keyword}`);
          return response;
        }
      } else {
        if (query.indexOf(keyword) !== -1 || keyword.indexOf(query) !== -1) {
          console.log(`[模糊包含] 命中分頁 [${sheetName}], 關鍵字: ${keyword}`);
          if (!fallbackCandidate) {
            fallbackCandidate = response;
          }
        }
      }
    }
  }

  return fallbackCandidate ? fallbackCandidate : DEFAULT_FALLBACK_MESSAGE;
}

// =================================================================
// 5. 呼叫 LINE Reply API 回傳文字
// =================================================================

function replyMessage(replyToken, replyText) {
  const url = 'https://api.line.me/v2/bot/message/reply';
  const payload = {
    replyToken: replyToken,
    messages: [{ type: 'text', text: replyText }]
  };

  const options = {
    method: 'post',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer ' + CHANNEL_ACCESS_TOKEN
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const responseCode = response.getResponseCode();
    if (responseCode !== 200) {
      console.error('LINE Reply API 異常: ' + response.getContentText());
    }
  } catch (err) {
    console.error('呼叫 LINE Reply API 發生錯誤: ' + err.toString());
  }
}

// 🔒 伺服器端核心驗證函式：阻絕未授權的修改與查詢
function verifyAdminAuth(password) {
  if (!ADMIN_PASSWORD || ADMIN_PASSWORD === 'YOUR_ADMIN_PASSWORD_HERE') {
    throw new Error('系統安全提示：請先於 Code.gs 第 14 行設定 ADMIN_PASSWORD 管理員密碼！');
  }
  if (!password || String(password) !== String(ADMIN_PASSWORD)) {
    throw new Error('存取遭拒：管理員密碼錯誤或登入已過期！');
  }
}

// 驗證管理員密碼
function apiVerifyAdmin(password) {
  verifyAdminAuth(password);
  return { success: true, message: '驗證通過' };
}

// 取得所有分類工作表清單 (需密碼)
function apiGetCategories(password) {
  verifyAdminAuth(password);
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  const sheets = spreadsheet.getSheets();
  const categories = [];

  for (let i = 0; i < sheets.length; i++) {
    const name = sheets[i].getName();
    if (EXCLUDED_SHEETS.indexOf(name) === -1) {
      categories.push({
        name: name,
        rowCount: Math.max(0, sheets[i].getLastRow() - 1)
      });
    }
  }

  // 若完全無有效分頁，自動初始化建立 FAQ 分頁
  if (categories.length === 0) {
    const defaultSheet = spreadsheet.insertSheet('FAQ');
    defaultSheet.appendRow(['關鍵字', '回覆內容', '比對模式']);
    categories.push({ name: 'FAQ', rowCount: 0 });
  }

  return {
    spreadsheetUrl: spreadsheet.getUrl(),
    categories: categories
  };
}

// 讀取指定工作表的所有問答項目 (需密碼)
function apiGetSheetData(password, sheetName) {
  verifyAdminAuth(password);
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) throw new Error('找不到工作表: ' + sheetName);

  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];

  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  const items = [];

  for (let i = 0; i < values.length; i++) {
    items.push({
      rowIndex: i + 2, // 試算表中的實際行數 (1-based)
      keyword: String(values[i][0] || ''),
      response: String(values[i][1] || ''),
      matchType: String(values[i][2] || 'contains')
    });
  }

  return items;
}

// 儲存或新增問答項目 (需密碼)
function apiSaveFaqItem(password, sheetName, item) {
  verifyAdminAuth(password);
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) throw new Error('找不到工作表: ' + sheetName);

  const keyword = String(item.keyword || '').trim();
  const response = String(item.response || '').trim();
  const matchType = String(item.matchType || 'contains').trim();

  if (!keyword || !response) {
    throw new Error('關鍵字與回覆內容不能為空白！');
  }

  if (item.rowIndex && item.rowIndex > 1) {
    // 編輯現有項目
    sheet.getRange(item.rowIndex, 1, 1, 3).setValues([[keyword, response, matchType]]);
    return { action: 'updated', rowIndex: item.rowIndex };
  } else {
    // 新增項目至最後一行
    sheet.appendRow([keyword, response, matchType]);
    return { action: 'created', rowIndex: sheet.getLastRow() };
  }
}

// 刪除問答項目 (需密碼)
function apiDeleteFaqItem(password, sheetName, rowIndex) {
  verifyAdminAuth(password);
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) throw new Error('找不到工作表: ' + sheetName);

  if (rowIndex > 1 && rowIndex <= sheet.getLastRow()) {
    sheet.deleteRow(rowIndex);
    return { success: true };
  }
  throw new Error('指定刪除的行號無效');
}

// 新增一個新的分類分頁 (需密碼)
function apiCreateCategory(password, newCategoryName) {
  verifyAdminAuth(password);
  const name = String(newCategoryName || '').trim();
  if (!name) throw new Error('分類名稱不可為空！');

  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  if (spreadsheet.getSheetByName(name)) {
    throw new Error('此分類分頁已存在！');
  }

  const newSheet = spreadsheet.insertSheet(name);
  newSheet.appendRow(['關鍵字', '回覆內容', '比對模式']);
  // 將標題列置中加粗
  newSheet.getRange('A1:C1').setFontWeight('bold').setBackground('#E6F4EA');
  return { success: true, name: name };
}

第五步：跨工作表檢索演算法與比對規則

訪客發送訊息：「請問信義門市的營業時間？」
               │
               ▼
 [階段一：完全相符比對 (Exact Match)]
 依序掃描所有工作表，尋找是否有關鍵字 100% 全等於提問句？
   ├─ 是 ──> 立即回傳該解答並結束 (優先權最高)
   └─ 否 ──> 進入階段二
               │

實務維運、排錯與安全性最佳實踐

Q1：在後台網頁新增資料後，LINE 機器人會即時生效嗎？

會！完全不需要重新部署。後台是直接對 Google Sheets 寫入，LINE 每次接收訪客對話皆即時讀取試算表儲存格。

Q2：在 GAS 修改了 ADMIN_PASSWORD 或 JavaScript 卻沒有生效？

這是 GAS 的版本快照機制所致。每次修改完 Code.gs 的密碼或程式碼，必須點選「部署」 > 「管理部署」 > 鉛筆圖示 (編輯) > 版本選擇「新版本」 > 點擊「部署」。

Q3：為什麼 GAS 部署時必須選擇「所有人 (Anyone)」，這樣安全嗎？

選擇「所有人」是為了讓 LINE 伺服器能在免 Google 帳號授權的狀態下，順利發送 Webhook 請求（doPost）。安全性完全不用擔心：所有涉及讀取、修改、刪除試算表資料的 RPC 函式皆已在伺服器端實施嚴格的 verifyAdminAuth 密碼檢查，外人即使知道網址或試圖竄改前端，伺服器端都會直接阻絕。
